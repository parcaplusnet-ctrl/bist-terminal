def hhmm(s):
    return s or "00:00"


def simdi_ist():
    return datetime.datetime.now(datetime.timezone.utc).replace(tzinfo=None) + datetime.timedelta(hours=3)


def yukle_radar(yol):
    spec = importlib.util.spec_from_file_location("radar_mod", yol)
    m = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(m)
    return m


def gunleri_oku(arsiv, bugun):
    G = {}
    for f in sorted(glob.glob(os.path.join(arsiv, "gozlem_bist_*.json"))):
        try:
            g = json.load(open(f, encoding="utf-8"))
            G[g["tarih"]] = g
        except Exception:
            continue
    return G


def xu(t):
    try:
        return float(((t.get("piyasa") or {}).get("xu100") or {}).get("v"))
    except Exception:
        return None


def gozlemler(R, G, bugun, simdi):
    """Tamamlanmış her gün için hisse gözlemleri (sabah özellikleri, endekse göre gün içi sonuç)."""
    obs, gunler = [], []
    tarihler = sorted(G)
    for i, gun in enumerate(tarihler):
        g = G[gun]
        T = [t for t in g.get("taramalar", []) if t.get("rows")]
        if len(T) < 2:
            continue
        t0, t1 = T[0], T[-1]
        if hhmm(t0["saat"]) > SABAH_EN_GEC or hhmm(t1["saat"]) < GUN_SONU_EN_ERKEN:
            continue                      # sabah taraması ya da gün sonu taraması yok: ölçülmez (ufuklar karışmasın)
        if gun == bugun and not (simdi.hour * 60 + simdi.minute >= 18 * 60 + 15):
            continue
        C = g["cols"]
        x0, x1 = xu(t0), xu(t1)
        A = {r[0]: r for r in t0["rows"]}
        Z = {r[0]: r for r in t1["rows"]}
        ix = {c: k for k, c in enumerate(C)}
        ham = []
        for kod, r in A.items():
            z = Z.get(kod)
            if not z or not r[1] or not z[1]:
                continue
            ch0 = r[ix["ch"]]
            if ch0 is None or abs(ch0) >= KILIT:
                continue
            ham.append((kod, r, z, (z[1] / r[1] - 1) * 100))
        if len(ham) < 100:
            continue
        ex = ((x1 / x0 - 1) * 100) if (x0 and x1) else st.median(h[3] for h in ham)
        for kod, r, z, ret in ham:
            F = set(R._ozellikler(r, C, "bist"))
            obs.append({"gun": gun, "kod": kod, "f": F, "ret": ret, "rel": ret - ex, "ex": ex,
                        "kap": z[ix["ch"]], "sab": r[ix["ch"]], "rv": r[ix["rv"]], "puan": r[ix["puan"]], "rsi": r[ix["rsi"]],
                        "sektor": r[ix["sektor"]], "fiyat0": r[1], "fiyat1": z[1], "likit": r[ix["likit_mn"]] if "likit_mn" in ix else None})
        gunler.append({"gun": gun, "ilk": t0["saat"], "son": t1["saat"], "n": len(ham), "endeks": round(ex, 2),
                       "medyan_hisse": round(st.median(h[3] for h in ham), 2)})
    return obs, gunler


def olc(idx, O):
    """İndeks kümesi için ölçüler."""
    n = len(idx)
    if n == 0:
        return None
    rel = [O[i]["rel"] for i in idx]
    yuk = sum(1 for v in rel if v >= ESIK_YUK) / n
    dus = sum(1 for v in rel if v <= ESIK_DUS) / n
    return {"n": n, "yuk": yuk, "dus": dus, "ort": sum(rel) / n, "med": st.median(rel)}


def z_oran(p, p0, n):
    if n == 0 or p0 <= 0 or p0 >= 1:
        return 0.0
    return (p - p0) / math.sqrt(p0 * (1 - p0) / n)


def tutarlilik(idx, O, yon, taban_gun):
    """Her günde (>=4 gözlem) örüntü ortalaması o günün tabanından yön doğrultusunda mı."""
    by = collections.defaultdict(list)
    for i in idx:
        by[O[i]["gun"]].append(O[i]["rel"])
    ok = say = 0
    for g, v in by.items():
        if len(v) < 4:
            continue
        say += 1
        fark = sum(v) / len(v) - taban_gun.get(g, 0)
        if (fark > 0 and yon > 0) or (fark < 0 and yon < 0):
            ok += 1
    return (ok / say if say else 0.0), say


def degerlendir(idx_all, O, kesif_gunleri, test_gunleri, taban_k, taban_i, taban_gun, ad_n_min):
    kes = [i for i in idx_all if O[i]["gun"] in kesif_gunleri]
    tst = [i for i in idx_all if O[i]["gun"] in test_gunleri]
    mk = olc(kes, O)
    if not mk or mk["n"] < ad_n_min:
        return None
    fark_yuk = mk["yuk"] - taban_k["yuk"]
    fark_dus = mk["dus"] - taban_k["dus"]
    fark_ort = mk["ort"] - taban_k["ort"]
    z_yuk = z_oran(mk["yuk"], taban_k["yuk"], mk["n"])
    z_dus = z_oran(mk["dus"], taban_k["dus"], mk["n"])
    if fark_ort >= ORT_FARK_MIN and z_yuk >= Z_MIN and fark_yuk > 0:
        yon, z, fark = 1, z_yuk, fark_yuk
    elif fark_ort <= -ORT_FARK_MIN and z_dus >= Z_MIN and fark_dus > 0:
        yon, z, fark = -1, z_dus, fark_dus
    else:
        return {"yon": 0, "n": mk["n"], "fark_ort": fark_ort, "fark_yuk": fark_yuk, "fark_dus": fark_dus, "z": max(z_yuk, z_dus),
                "yuk": mk["yuk"], "dus": mk["dus"], "ort": mk["ort"], "gecti_kesif": False}
    tut, tgun = tutarlilik(kes, O, yon, taban_gun)
    gecti = tut >= TUTARLI_MIN and tgun >= 3
    r = {"yon": yon, "n": mk["n"], "yuk": mk["yuk"], "dus": mk["dus"], "ort": mk["ort"], "fark_ort": fark_ort, "fark_yuk": fark_yuk,
         "fark_dus": fark_dus, "z": z, "tutarlilik": tut, "gun_sayisi": tgun, "gecti_kesif": gecti}
    mt = olc(tst, O)
    if mt:
        r["n_ileri"] = mt["n"]
        r["ileri_fark_ort"] = mt["ort"] - taban_i["ort"]
        r["ileri_fark_yuk"] = mt["yuk"] - taban_i["yuk"]
        r["ileri_fark_dus"] = mt["dus"] - taban_i["dus"]
    else:
        r["n_ileri"] = 0
    return r


def durum_belirle(r, gun_sayisi, test_var):
    if r is None:
        return None
    if gun_sayisi < MIN_GUN:
        return "Veri biriktiriliyor"
    if not r.get("gecti_kesif"):
        return "Kanıt yok"
    if not test_var or r.get("n_ileri", 0) < ILERI_N:
        return "İleri test bekliyor"
    yon = r["yon"]
    ileri_ort = r["ileri_fark_ort"] * yon
    ileri_oran = (r["ileri_fark_yuk"] if yon > 0 else r["ileri_fark_dus"])
    if ileri_ort >= ILERI_ORAN * abs(r["fark_ort"]) and ileri_oran > 0:
        return "Onay bekliyor"
    return "Elendi (ileri test)"

# ---- Sürpriz yükselenler (19.11, kullanıcı kararı): gün içinde YATAY başlayıp +%5 ve üstüne çıkan hisseler ----
SURPRIZ_YATAY = 1.0        # sabah taramasında |gün değişimi| bu değerin altındaysa "yatay başladı"
SURPRIZ_ESIK = 5.0         # gün içi en yüksek gün değişimi bu değere ulaşırsa "sürpriz yükselen"
SURPRIZ_BASLA = 2.0        # hareketin başlangıcı: sabaha göre ilk +2 puanlık sıçrama yapan tarama
SURPRIZ_LIKIT = 20.0       # günlük ciro (mn TL) altı hisseler dışlanır (sığ hisse sıçramaları gürültüdür)
SURPRIZ_RV = 2.0           # başlangıç taramasında göreli hacim bu değerin üstündeyse "hacim patlaması"


def _dk(s):
    try:
        h, m = str(s).split(":")[:2]
        return int(h) * 60 + int(m)
    except Exception:
        return 0


def _surpriz_gun(R, g, gun, simdi, bugun):
    """Bir günün sürpriz yükselen kayıtları. Dönen: (özet, olaylar) ya da None. Bugün için canlı (gün devam ediyor)."""
    T = [t for t in g.get("taramalar", []) if t.get("rows")]
    if len(T) < 2 or hhmm(T[0]["saat"]) > SABAH_EN_GEC:
        return None
    C = g["cols"]; ix = {c: k for k, c in enumerate(C)}
    devam = hhmm(T[-1]["saat"]) < GUN_SONU_EN_ERKEN
    if devam and gun != bugun:
        return None                     # eski ve eksik gün: ölçülmez
    S = [{r[0]: r for r in t["rows"]} for t in T]
    sek = collections.defaultdict(list)
    for kod, r in S[0].items():
        sek[r[ix["sektor"]]].append(kod)
    yatay = []
    for kod, r in S[0].items():
        ch0 = r[ix["ch"]]
        lk = r[ix["likit_mn"]] if "likit_mn" in ix else None
        if ch0 is None or abs(ch0) > SURPRIZ_YATAY or not r[1] or (lk is not None and lk < SURPRIZ_LIKIT):
            continue
        yatay.append(kod)
    olaylar = []
    for kod in yatay:
        seri = []
        for i, s in enumerate(S):
            r = s.get(kod)
            seri.append((T[i]["saat"], r[ix["ch"]], r[ix["rv"]], r[1]) if r and r[ix["ch"]] is not None and r[1] else None)
        if seri[0] is None:
            continue
        ch0 = seri[0][1]
        gecer = [(i, x) for i, x in enumerate(seri) if x is not None and i > 0]
        if not gecer:
            continue
        zi, zx = max(gecer, key=lambda q: q[1][1])
        if zx[1] < SURPRIZ_ESIK:
            continue
        bi = next((i for i, x in gecer if x[1] >= ch0 + SURPRIZ_BASLA), zi)
        bx = seri[bi]
        son = next((x for x in reversed(seri) if x is not None), None)
        sec = S[0][kod][ix["sektor"]]
        es = [k for k in sek.get(sec, []) if k != kod and S[bi].get(k) and S[0].get(k)]
        farklar = [S[bi][k][ix["ch"]] - S[0][k][ix["ch"]] for k in es if S[bi][k][ix["ch"]] is not None and S[0][k][ix["ch"]] is not None]
        es_ort = round(sum(farklar) / len(farklar), 2) if farklar else None
        es_yuk = sum(1 for k in es if S[bi][k][ix["ch"]] is not None and S[bi][k][ix["ch"]] >= 3.0)
        x0, xb = xu(T[0]), xu(T[bi])
        ex = round((xb / x0 - 1) * 100, 2) if (x0 and xb) else None
        F = set(R._ozellikler(S[0][kod], C, "bist"))
        ozl = [f for f in sorted(F) if f.split("=")[0] in ("hacim", "gün", "puan", "RSI G", "1 ay zirve", "hafta")]
        rv0, rvb = seri[0][2], bx[2]
        kat = ("sektör hareketi" if (es_ort is not None and es_ort >= 1.0 and len(farklar) >= 5)
               else "hacim patlaması" if (rvb is not None and rvb >= SURPRIZ_RV)
               else "piyasa geneli" if (ex is not None and ex >= 1.0)
               else "açıklanamayan")
        olaylar.append({"gun": gun, "kod": kod, "sektor": sec, "sabah": round(ch0, 2), "baslangic": bx[0], "baslangic_ch": round(bx[1], 2), "zirve": round(zx[1], 2), "zirve_saat": zx[0],
                        "son": round(son[1], 2) if son else None, "son_saat": son[0] if son else None, "rv_sabah": rv0, "rv_baslangic": rvb,
                        "es_sektor_ort": es_ort, "es_sektor_yukselen": es_yuk, "endeks_degisim": ex, "fiyat_baslangic": bx[3], "fiyat_zirve": zx[3], "fiyat_son": son[3] if son else None,
                        "kategori": kat, "haber": None, "haber_saat": None, "ozellik": ozl})
    olaylar.sort(key=lambda o: -o["zirve"])
    return {"gun": gun, "devam": devam, "yatay_n": len(yatay), "olay_n": len(olaylar), "ilk": T[0]["saat"], "son": T[-1]["saat"], "n_tarama": len(T)}, olaylar, (S, T, ix, yatay, sek)


def _haber_ekle(R, olaylar, kayit, simdi):
    """Her olay için hareketten önce (önceki gün 17:00'dan başlangıç taramasına +30 dk) yayınlanmış haber başlığı var mı? Sonuç kalıcı önbellekte (ağ yoksa atlanır)."""
    H = kayit.setdefault("surpriz_haber", {})
    getirilen = 0
    for o in olaylar:
        k = f"{o['gun']}|{o['kod']}"
        if (k not in H or o["gun"] == simdi.strftime("%Y-%m-%d")) and getirilen < 25:
            try:
                j = R.http(f"https://news-headlines.tradingview.com/v2/headlines?client=web&lang=tr&symbol=BIST:{o['kod']}&streaming=false", deneme=2, bekle=1.0)
                H[k] = [{"ts": i["published"], "baslik": str(i["title"])[:140]} for i in (j.get("items") or [])[:12]]
                getirilen += 1
                import time as _t
                _t.sleep(.3)
            except Exception:
                continue
        L = H.get(k)
        if L is None:
            continue
        g = datetime.datetime.strptime(o["gun"], "%Y-%m-%d")
        lo = (g - datetime.timedelta(days=1)).replace(hour=17) - datetime.timedelta(hours=3)          # İstanbul -> UTC
        hh, mm = o["baslangic"].split(":")[:2]
        hi = g.replace(hour=int(hh), minute=int(mm)) - datetime.timedelta(hours=3) + datetime.timedelta(minutes=30)
        lo_ts = lo.replace(tzinfo=datetime.timezone.utc).timestamp(); hi_ts = hi.replace(tzinfo=datetime.timezone.utc).timestamp()
        uy = [x for x in L if lo_ts <= x["ts"] <= hi_ts]
        if uy:
            x = max(uy, key=lambda z: z["ts"])
            o["haber"] = x["baslik"]
            o["haber_saat"] = (datetime.datetime.fromtimestamp(x["ts"], datetime.timezone.utc) + datetime.timedelta(hours=3)).strftime("%d.%m %H:%M")
            o["kategori"] = "haber"
        o["haber_kontrol"] = True
    return olaylar


def surpriz(R, G, bugun, simdi, kayit):
    """HL['surpriz']: bugünkü (canlı) ve en son tamamlanmış günün sürpriz yükselenleri, taban oranlar, saat ve neden dağılımı, erken işaret adayları (yalnızca ölçüm; kural değil)."""
    gunler, tum, erken_veri = [], [], []
    for gun in sorted(G):
        r = _surpriz_gun(R, G[gun], gun, simdi, bugun)
        if not r:
            continue
        ozet, olaylar, (S, T, ix, yatay, sek) = r
        gunler.append(ozet); tum.append((ozet, olaylar))
        if not ozet["devam"] and len(T) >= 3:
            # erken işaret: ikinci taramadaki (ilk saat sonrası) durum -> sonraki taramalarda +%5'e ulaşma
            ev = {o["kod"] for o in olaylar}
            for kod in yatay:
                if kod not in S[1] or kod not in S[0]:
                    continue
                a0, a1 = S[0][kod], S[1][kod]
                if a1[ix["ch"]] is None or a0[ix["ch"]] is None or not a1[1] or a1[ix["ch"]] >= SURPRIZ_ESIK:
                    continue
                sec = a0[ix["sektor"]]
                es = [k for k in sek.get(sec, []) if k != kod and k in S[1] and S[1][k][ix["ch"]] is not None and S[0][k][ix["ch"]] is not None]
                esf = [S[1][k][ix["ch"]] - S[0][k][ix["ch"]] for k in es]
                es_ort = (sum(esf) / len(esf)) if esf else None
                sonra = [S[i][kod][ix["ch"]] for i in range(2, len(S)) if kod in S[i] and S[i][kod][ix["ch"]] is not None]
                if not sonra:
                    continue
                kap = S[-1][kod][ix["ch"]] if kod in S[-1] else None
                erken_veri.append({"gun": ozet["gun"], "d1": a1[ix["ch"]] - a0[ix["ch"]], "rv1": a1[ix["rv"]], "es": es_ort, "hit": max(sonra) >= SURPRIZ_ESIK,
                                   "getiri": ((S[-1][kod][1] / a1[1] - 1) * 100) if (kod in S[-1] and S[-1][kod][1]) else None})
    HS = {"tanim": f"Sabah taramasında |gün değişimi| ≤ %{SURPRIZ_YATAY:g} (yatay başlayan, günlük ciro ≥ {SURPRIZ_LIKIT:g} mn TL) olup gün içinde en az +%{SURPRIZ_ESIK:g}'e çıkan hisse. Başlangıç = sabaha göre ilk +{SURPRIZ_BASLA:g} puanlık sıçrama yapan tarama. Neden kategorisi otomatiktir: haber (önceki gün 17:00 ile başlangıç +30 dk arasında yayın), sektör hareketi (aynı sektörün diğer hisseleri sabahtan başlangıca kadar ortalama en az +1 puan yükseldi), hacim patlaması (başlangıçta göreli hacim ≥ {SURPRIZ_RV:g}x), piyasa geneli (endeks sabaha göre ≥ +1), açıklanamayan. Yalnızca ölçüm; seçime etkisi yoktur.",
          "gun_sayisi": len([g for g in gunler if not g["devam"]]), "gunler": [], "bugun": None, "son_gun": None, "saat_dagilimi": [], "kategoriler": [], "erken_isaret": [], "taban": None}
    if not tum:
        HS["not"] = "Ölçülebilir gün yok (sabah taraması <= 11:30 ve en az 2 tarama gerekir)."
        return HS
    # haber kontrolü: yalnızca bugün ve en son tamamlanmış gün (eskiler önbellekten)
    bugun_k = next((x for x in tum if x[0]["gun"] == bugun), None)
    son_k = next((x for x in reversed(tum) if not x[0]["devam"]), None)
    for k in (bugun_k, son_k):
        if k:
            _haber_ekle(R, k[1], kayit, simdi)
    H = kayit.get("surpriz_haber") or {}
    for ozet, olaylar in tum:
        if ozet["gun"] not in (bugun_k[0]["gun"] if bugun_k else None, son_k[0]["gun"] if son_k else None):
            _haber_ekle_onbellek = [o for o in olaylar if f"{o['gun']}|{o['kod']}" in H]
            if _haber_ekle_onbellek:
                _haber_ekle(R, _haber_ekle_onbellek, kayit, simdi)
    son_gunler = {o["gun"] for o, _ in tum[-10:]}
    kayit["surpriz_haber"] = {k: v for k, v in (kayit.get("surpriz_haber") or {}).items() if k.split("|")[0] in son_gunler}
    tam = [x for x in tum if not x[0]["devam"]]
    HS["gunler"] = [{"gun": o["gun"], "yatay_n": o["yatay_n"], "olay_n": o["olay_n"], "oran": round(o["olay_n"] / o["yatay_n"], 3) if o["yatay_n"] else None, "devam": o["devam"]} for o, _ in tum[-10:]]
    ty, to = sum(o["yatay_n"] for o, _ in tam), sum(o["olay_n"] for o, _ in tam)
    HS["taban"] = {"yatay_n": ty, "olay_n": to, "oran": round(to / ty, 4) if ty else None}
    def liste(k):
        o, ol = k
        return {"gun": o["gun"], "devam": o["devam"], "ilk": o["ilk"], "son": o["son"], "yatay_n": o["yatay_n"], "olay_n": o["olay_n"], "liste": ol[:30]}
    HS["bugun"] = liste(bugun_k) if bugun_k else None
    HS["son_gun"] = liste(son_k) if son_k else None
    tum_ol = [o for _, ol in tam for o in ol]
    sd = collections.Counter(o["baslangic"][:2] + ":00" for o in tum_ol)
    HS["saat_dagilimi"] = [{"saat": s, "n": n} for s, n in sorted(sd.items())]
    kd = collections.defaultdict(list)
    for o in tum_ol:
        kd[o["kategori"]].append(o)
    HS["kategoriler"] = sorted([{"kategori": k, "n": len(v), "pay": round(len(v) / len(tum_ol), 3), "ort_zirve": round(sum(x["zirve"] for x in v) / len(v), 2),
                                 "ort_son": round(sum(x["son"] for x in v if x["son"] is not None) / max(1, sum(1 for x in v if x["son"] is not None)), 2)} for k, v in kd.items()], key=lambda r: -r["n"])
    taban_oran = (to / ty) if ty else None
    kosullar = [("İlk saat +2 puan ve üstü", lambda e: e["d1"] >= 2.0), ("İlk saat +2 puan ve göreli hacim ≥ 1,5x", lambda e: e["d1"] >= 2.0 and (e["rv1"] or 0) >= 1.5),
                ("İlk saat +1 puan ve göreli hacim ≥ 2x", lambda e: e["d1"] >= 1.0 and (e["rv1"] or 0) >= 2.0), ("Göreli hacim ≥ 2x (fiyattan bağımsız)", lambda e: (e["rv1"] or 0) >= 2.0),
                ("Sektör arkadaşları ilk saatte ort. +1 puan ve üstü", lambda e: e["es"] is not None and e["es"] >= 1.0)]
    ng_ = len({e["gun"] for e in erken_veri})
    for ad, f in kosullar:
        S_ = [e for e in erken_veri if f(e)]
        n = len(S_); h = sum(e["hit"] for e in S_)
        gl = [e["getiri"] for e in S_ if e["getiri"] is not None]
        oran = round(h / n, 3) if n else None
        durum = "Veri biriktiriliyor" if (ng_ < MIN_GUN or n < 30) else ("taban üstü" if (oran is not None and taban_oran and oran >= 1.5 * taban_oran) else "taban civarı")
        HS["erken_isaret"].append({"kosul": ad, "n": n, "isabet": oran, "kat": round(oran / taban_oran, 1) if (oran is not None and taban_oran) else None,
                                   "ort_kapanis_getiri": round(sum(gl) / len(gl), 2) if gl else None, "durum": durum})
    HS["erken_isaret_not"] = f"Erken işaret = ikinci taramadaki (ilk saat sonrası) durum; isabet = sonraki taramalarda +%{SURPRIZ_ESIK:g}'e ulaşma. {ng_} gün, yatay başlayan hisseler. Kanıt kapısı: en az {MIN_GUN} gün ve 30 örnek; geçse bile yalnızca öneri, seçime kullanıcı onayı olmadan girmez."
    return HS


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--motor", default="radar.py")
    ap.add_argument("--arsiv", default="arsiv")
    ap.add_argument("--veri", default="veri.json")
    ap.add_argument("--kayit", default="lab/hareket.json")
    ap.add_argument("--rules", default="rules.json")
    ap.add_argument("--neden", default="hareket_neden.json")
    ap.add_argument("--bugun", default=None)
    a = ap.parse_args()
    simdi = simdi_ist()
    bugun = a.bugun or simdi.strftime("%Y-%m-%d")
    R = yukle_radar(a.motor)
    G = gunleri_oku(a.arsiv, bugun)
    O, gunler = gozlemler(R, G, bugun, simdi)
    V = json.load(open(a.veri, encoding="utf-8"))
    try:
        rules = json.load(open(a.rules, encoding="utf-8"))
    except Exception:
        rules = {}
    onayli = {k["id"]: k for k in rules.get("hareket_kurallari", [])}
    kayit = {}
    if os.path.exists(a.kayit):
        try:
            kayit = json.load(open(a.kayit, encoding="utf-8"))
        except Exception:
            kayit = {}
    zaman = simdi.strftime("%d.%m %H:%M")
    gun_adlari = [g["gun"] for g in gunler]
    ng = len(gun_adlari)
    HL = {"surum": VERSIYON, "guncelleme": zaman, "gun_sayisi": ng, "gunler": gunler[-10:], "gozlem": len(O),
          "kurallar": {"yukselen": f"endeksten >= +{ESIK_YUK:g} puan fazla", "dusen": f"endeksten <= {ESIK_DUS:g} puan az", "min_gun": MIN_GUN,
                       "kesif_ornek": f"tek {KESIF_N1}, çift {KESIF_N2}", "z": Z_MIN, "ort_fark": ORT_FARK_MIN, "tutarlilik": TUTARLI_MIN,
                       "ileri_ornek": ILERI_N, "ileri_etki": ILERI_ORAN,
                       "not": "Sabah taraması (<= 11:30) -> gün sonu taraması (>= 17:45), endekse göre. Kilitli (|sabah gün| >= %9,5) hisseler dışlanır. Önerilen örüntü ancak kullanıcı onayıyla seçime girer."},
          "oruntular": [], "sistemler": [], "dun": None, "onayli": []}
    try:
        HL["surpriz"] = surpriz(R, G, bugun, simdi, kayit)
    except Exception as e:
        print("UYARI: sürpriz yükselen ölçümü", repr(e))
    if ng == 0:
        HL["not"] = "Ölçülebilir tamamlanmış gün yok (sabah ve gün sonu taraması gerekir)."
        V["hareket_lab"] = HL
        json.dump(V, open(a.veri, "w", encoding="utf-8"), ensure_ascii=False, separators=(",", ":"))
        print("HAREKET: ölçülebilir gün yok")
        return
    # keşif / ileri test günleri
    K = max(2, round(ng * 0.25)) if ng >= MIN_GUN else 0
    test_g = set(gun_adlari[-K:]) if K else set()
    kesif_g = set(gun_adlari) - test_g
    idx_all = list(range(len(O)))
    kes_idx = [i for i in idx_all if O[i]["gun"] in kesif_g]
    tst_idx = [i for i in idx_all if O[i]["gun"] in test_g]
    taban_k = olc(kes_idx, O)
    taban_i = olc(tst_idx, O) or taban_k
    taban_gun = {}
    for g in gun_adlari:
        v = [O[i]["rel"] for i in idx_all if O[i]["gun"] == g]
        taban_gun[g] = sum(v) / len(v) if v else 0.0
    HL["taban"] = {"n": taban_k["n"], "yukselen": round(taban_k["yuk"], 3), "dusen": round(taban_k["dus"], 3), "ort_rel": round(taban_k["ort"], 2),
                   "kesif_gunleri": sorted(kesif_g), "ileri_gunleri": sorted(test_g)}
    # özellik -> gözlem indeksleri
    harita = collections.defaultdict(list)
    for i in idx_all:
        for f in O[i]["f"]:
            harita[f].append(i)
    aile = lambda f: f.split("=")[0]
    tekler = {f: v for f, v in harita.items() if len(v) >= 30}
    sonuc = []
    for f, v in tekler.items():
        r = degerlendir(v, O, kesif_g, test_g, taban_k, taban_i, taban_gun, KESIF_N1)
        if r:
            sonuc.append(([f], r))
    # çiftler: yalnızca kendi başına kanıt/eğilim taşıyan ya da yeterince kalabalık özelliklerden, farklı ailelerden
    adaylar = [f for f, v in tekler.items() if len(v) >= 120 and not f.startswith("sektör=") and not f.startswith("sinyal=")]
    kume = {f: set(tekler[f]) for f in adaylar}
    for f1, f2 in itertools.combinations(sorted(adaylar), 2):
        if aile(f1) == aile(f2):
            continue
        ortak = sorted(kume[f1] & kume[f2])
        if len(ortak) < KESIF_N2:
            continue
        r = degerlendir(ortak, O, kesif_g, test_g, taban_k, taban_i, taban_gun, KESIF_N2)
        if r and r["yon"]:
            sonuc.append(([f1, f2], r))
    # iç içe örüntüleri ele: tek özellik zaten aynı yönde aynı etkiyi taşıyorsa çift gereksiz
    tek_yon = {t[0]: r for t, r in sonuc if len(t) == 1 and r["yon"] and r.get("gecti_kesif")}
    liste = []
    for feats, r in sonuc:
        if not r["yon"]:
            continue
        if len(feats) == 2 and any(f in tek_yon and tek_yon[f]["yon"] == r["yon"] and abs(tek_yon[f]["fark_ort"]) >= abs(r["fark_ort"]) * 0.9 for f in feats):
            continue
        d = durum_belirle(r, ng, bool(test_g))
        pid = "|".join(sorted(feats))
        liste.append((pid, feats, r, d))
    # sıralama: kanıt gücü (z), yön
    liste.sort(key=lambda x: -x[2]["z"])
    # kalıcı kayıt: durum geçmişi
    Kk = kayit.setdefault("oruntular", {})
    for pid, feats, r, d in liste:
        e = Kk.setdefault(pid, {"aciklama": aciklama(feats), "ilk_gorulme": zaman, "durum_gecmisi": []})
        if not e["durum_gecmisi"] or e["durum_gecmisi"][-1]["durum"] != d:
            e["durum_gecmisi"].append({"zaman": zaman, "durum": d, "gun_sayisi": ng})
            e["durum_gecmisi"] = e["durum_gecmisi"][-12:]
    for pid, feats, r, d in liste[:60]:
        o = {"id": pid, "aciklama": aciklama(feats), "yon": "bonus" if r["yon"] > 0 else "eleme", "durum": d, "n": r["n"],
             "yuk_oran": round(r["yuk"], 3), "dus_oran": round(r["dus"], 3), "ort_rel": round(r["ort"], 2),
             "fark_yuk": round(r["fark_yuk"], 3), "fark_dus": round(r["fark_dus"], 3), "fark_ort": round(r["fark_ort"], 2), "z": round(r["z"], 1),
             "tutarlilik": round(r.get("tutarlilik", 0), 2), "n_ileri": r.get("n_ileri", 0),
             "ileri_fark_ort": round(r["ileri_fark_ort"], 2) if r.get("ileri_fark_ort") is not None else None,
             "ilk_gorulme": Kk[pid]["ilk_gorulme"], "onayli": pid in onayli}
        HL["oruntular"].append(o)
    HL["ozet"] = {"toplam": len(liste), "onay_bekleyen": sum(1 for x in liste if x[3] == "Onay bekliyor"),
                  "ileri_test_bekleyen": sum(1 for x in liste if x[3] == "İleri test bekliyor"),
                  "veri_biriktiriliyor": sum(1 for x in liste if x[3] == "Veri biriktiriliyor"),
                  "elenen": sum(1 for x in liste if x[3] and x[3].startswith("Elendi")),
                  "durum": ("Veri biriktiriliyor: %d/%d gün; kanıt kapıları %d günden sonra işler" % (ng, MIN_GUN, MIN_GUN)) if ng < MIN_GUN else "Kanıt kapıları açık"}
    # çalışan sistemlerin puanlanması (her gün aynı yöntem)
    sis = list(SISTEMLER)
    for kid, k in onayli.items():
        sis.append({"id": "kural:" + kid, "ad": "Onaylı kural: " + k.get("aciklama", kid), "all": k.get("ozellik") or kid.split("|"), "yon": k.get("yon")})
    for s in sis:
        def eslesir(F, s=s):
            if "all" in s:
                return all(x in F for x in s["all"])
            return any(x in F for x in s["any"])
        gunluk = []
        tum = []
        for g in gun_adlari:
            ids = [i for i in idx_all if O[i]["gun"] == g and eslesir(O[i]["f"])]
            m = olc(ids, O)
            if m:
                gunluk.append({"gun": g, "n": m["n"], "ort_rel": round(m["ort"], 2), "yuk": round(m["yuk"], 3), "dus": round(m["dus"], 3),
                               "taban": round(taban_gun[g], 2)})
            tum += ids
        mm = olc(tum, O)
        if mm:
            top = olc(idx_all, O)
            HL["sistemler"].append({"id": s["id"], "ad": s["ad"], "n": mm["n"], "gun": len(gunluk), "ort_rel": round(mm["ort"], 2),
                                    "yuk": round(mm["yuk"], 3), "dus": round(mm["dus"], 3), "taban_ort": round(top["ort"], 2),
                                    "taban_yuk": round(top["yuk"], 3), "taban_dus": round(top["dus"], 3), "fark_ort": round(mm["ort"] - top["ort"], 2),
                                    "gunluk": gunluk[-10:], "yon": s.get("yon")})
    # kasa seçimlerinin gerçek sonuçları (kalite kapısı)
    H = [h for h in V.get("history", []) if h.get("sonuc") in ("Başarılı", "Stop", "Takip Süresi Doldu")]
    if H:
        b = sum(1 for h in H if h["sonuc"] == "Başarılı")
        sp = sum(1 for h in H if h["sonuc"] == "Stop")
        HL["sistemler"].append({"id": "secim", "ad": "Radar seçimleri (sonuçlanan kayıt)", "n": len(H), "gun": None, "ort_rel": None,
                                "yuk": round(b / len(H), 3), "dus": round(sp / len(H), 3), "taban_ort": None, "taban_yuk": None, "taban_dus": None,
                                "fark_ort": None, "gunluk": [], "yon": None})
    # dünün (en son tamamlanmış günün) hareketleri
    son = gun_adlari[-1]
    O_son = [o for o in O if o["gun"] == son]
    neden = dict(kayit.get("nedenler") or {})          # nedenler kalıcı kayıtta taşınır (saatlik görev de görür): {kod: {tarih, neden}}
    try:
        neden.update(json.load(open(a.neden, encoding="utf-8")))
    except Exception:
        pass
    kayit["nedenler"] = {k: v for k, v in neden.items() if isinstance(v, dict) and v.get("tarih") in gun_adlari[-3:]}
    O_son.sort(key=lambda o: o["rel"])

    def ozet(o):
        sec = [f for f in sorted(o["f"]) if f.split("=")[0] in ("hacim", "gün", "puan", "RSI G", "1 ay zirve", "hafta") ]
        return {"kod": o["kod"], "sektor": o["sektor"], "rel": round(o["rel"], 1), "gun_ici": round(o["ret"], 1), "sabah_degisim": o["sab"],
                "kapanis_degisim": o["kap"], "rv": o["rv"], "puan": o["puan"], "rsi": o["rsi"], "ozellik": sec,
                "neden": (neden.get(o["kod"]) or {}).get("neden") if (neden.get(o["kod"]) or {}).get("tarih") == son else None}
    yuk_l, dus_l = [o for o in O_son if o["rel"] >= ESIK_YUK], [o for o in O_son if o["rel"] <= ESIK_DUS]

    def ortak(grup):
        if len(grup) < 8:
            return []
        out = []
        sayac = collections.Counter(f for o in grup for f in o["f"])
        gen = collections.Counter(f for o in O_son for f in o["f"])
        for f, c in sayac.items():
            if c < 5 or f.startswith("sektör="):
                continue
            pay_g, pay_t = c / len(grup), gen[f] / len(O_son)
            if pay_t > 0 and pay_g / pay_t >= 1.5:
                out.append({"ozellik": f, "grupta": round(pay_g, 2), "tabanda": round(pay_t, 2), "kat": round(pay_g / pay_t, 1), "n": c})
        out.sort(key=lambda x: -x["kat"] * math.sqrt(x["n"]))
        return out[:8]
    HL["dun"] = {"gun": son, "endeks": [g for g in gunler if g["gun"] == son][0]["endeks"], "yukselen_sayi": len(yuk_l), "dusen_sayi": len(dus_l),
                 "gozlem": len(O_son), "yukselenler": [ozet(o) for o in O_son[::-1][:12]], "dusenler": [ozet(o) for o in O_son[:12]],
                 "yukselen_ortak": ortak(yuk_l), "dusen_ortak": ortak(dus_l),
                 "uyari": "Tek günlük gözlem kanıt değildir; örüntü tablosundaki kapılardan geçmeden seçime etkisi yoktur."}
    HL["onayli"] = [{"id": k, "aciklama": v.get("aciklama"), "yon": v.get("yon"), "onay": v.get("onay_tarihi")} for k, v in onayli.items()]
    V["hareket_lab"] = HL
    json.dump(V, open(a.veri, "w", encoding="utf-8"), ensure_ascii=False, separators=(",", ":"))
    kayit["guncelleme"] = zaman
    kayit["gun_sayisi"] = ng
    os.makedirs(os.path.dirname(a.kayit) or ".", exist_ok=True)
    json.dump(kayit, open(a.kayit, "w", encoding="utf-8"), ensure_ascii=False, indent=1)
    o = HL["ozet"]
    print(f"HAREKET: {ng} gün, {len(O)} gözlem · örüntü {o['toplam']} (onay bekleyen {o['onay_bekleyen']}, ileri test bekleyen {o['ileri_test_bekleyen']}, "
          f"elenen {o['elenen']}) · {o['durum']}")
    print(f"HAREKET DÜN ({son}): endeks {HL['dun']['endeks']:+.2f}% · endekse göre yükselen {len(yuk_l)}, düşen {len(dus_l)}")
    for s in HL["sistemler"]:
        if s.get("ort_rel") is not None:
            print(f"  SİSTEM {s['ad']}: n={s['n']} ort {s['ort_rel']:+.2f} (taban {s['taban_ort']:+.2f}) +3↑ %{s['yuk']*100:.0f} (taban %{s['taban_yuk']*100:.0f}) -3↓ %{s['dus']*100:.0f} (taban %{s['taban_dus']*100:.0f})")


if __name__ == "__main__":
    main()
